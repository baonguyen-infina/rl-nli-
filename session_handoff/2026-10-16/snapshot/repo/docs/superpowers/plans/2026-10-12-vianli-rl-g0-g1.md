## Global Constraints

- Never put `uid` / `intended_label` into prompts or features.
- Do not touch the `test` split (no reading, no counting) before G4; reading it requires `allow_test=True` AND env `G4_UNLOCK=1`.
- One parser only: `formats/parser.py` (used by reward, eval, teacher filter).
- Every run goes through `scripts/run.py --run <ID> --seed <S>` and writes to `results/<run_id>_<model>_s<seed>/`.
- Hyperparameters only in `configs/base.yaml`; per-run differences only in `configs/matrix.py`.
- Labels: `entailment | neutral | contradiction`. Prompts and premise/hypothesis in Vietnamese.
- Reward: `R = r_acc + 0.2*r_fmt + 0.3*r_acc*r_struct`; reward weights `[1.0, 0.2, 0.3]`; reward func names `r_acc`, `r_fmt`, `r_acc_struct`.
- Align: 1–6 lines, `missing` goes with `∅`; rule conflict → contradiction, else missing → neutral, else entailment; coverage ≥ 0.5.
- Rollout: temperature 1.0, top_p 1.0, no top_k/presence penalty. Eval: greedy.
- Main model `Qwen/Qwen3.5-2B` (non-thinking, language part only; vLLM `language_model_only=True`). `trl==1.15.0`.
- Run all tests from repo root with `python -m pytest`. Heavy libs (`torch`, `vllm`, `trl`, `transformers`, `peft`) imported lazily inside functions only.
- Output files per run: `config.yaml`, `train_log.csv`, `predictions_<split>.jsonl`, `metrics_<split>.json`, `cost.json`.
- Values marked [đề xuất] in the spec (hyperparameters) live only in `configs/base.yaml`.

```python
"""Chuẩn hoá NFC + khoảng trắng, khoá so khớp (casefold) cho grounding, tách âm tiết để tính coverage.
"""
import re
import unicodedata

_WS = re.compile(r"\s+")
_TOKEN = re.compile(r"\w+", re.UNICODE)


def normalize(s: str) -> str:
    return _WS.sub(" ", unicodedata.normalize("NFC", s)).strip()


def match_key(s: str) -> str:
    return normalize(s).casefold()


def syllables(s: str) -> list[str]:
    return _TOKEN.findall(match_key(s))
```

```python
"""Đường dẫn (artifacts/, results/, configs/runs/), đọc/ghi JSONL/JSON, đặt tên run `<run_id>_<model>_s<seed>`.
"""
import csv
import json
from pathlib import Path

import yaml

ROOT = Path(__file__).resolve().parents[1]
ARTIFACTS = ROOT / "artifacts"
RESULTS = ROOT / "results"
RUN_CONFIGS = ROOT / "configs" / "runs"


def read_jsonl(path) -> list[dict]:
    with open(path, encoding="utf-8") as f:
        return [json.loads(line) for line in f if line.strip()]


def write_jsonl(path, rows) -> None:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        for r in rows:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")


def read_json(path):
    with open(path, encoding="utf-8") as f:
        return json.load(f)


def write_json(path, obj) -> None:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", encoding="utf-8") as f:
        json.dump(obj, f, ensure_ascii=False, indent=2)


def run_name(run_id: str, model: str, seed: int) -> str:
    return f"{run_id}_{model.split('/')[-1]}_s{seed}"


def run_dir(run_id: str, model: str, seed: int) -> Path:
    return RESULTS / run_name(run_id, model, seed)


def deep_merge(a: dict, b: dict) -> dict:
    out = {k: (dict(v) if isinstance(v, dict) else v) for k, v in a.items()}
    for k, v in b.items():
        if isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k] = deep_merge(out[k], v)
        else:
            out[k] = v
    return out


def load_config(run_id: str) -> dict:
    with open(ROOT / "configs" / "base.yaml", encoding="utf-8") as f:
        base = yaml.safe_load(f) or {}
    with open(RUN_CONFIGS / f"{run_id}.yaml", encoding="utf-8") as f:
        run = yaml.safe_load(f) or {}
    return deep_merge(base, run)


def write_train_log(log_history: list[dict], path) -> None:
    fields: list[str] = []
    for row in log_history:
        for k in row:
            if k not in fields:
                fields.append(k)
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=fields)
        w.writeheader()
        for row in log_history:
            w.writerow(row)
```

```python
"""Parse dòng align, luật tổng hợp nhãn (conflict > missing > match), kiểm grounding, coverage >= 0.5, struct_ok.
"""
import re
from dataclasses import dataclass

from common.text import match_key, syllables

LINE_RE = re.compile(
    r'^H(\d+):\s*"(.+?)"\s*<->\s*(?:"(.+)"|∅)\s*\|\s*(match|missing|conflict)$'
)
COVERAGE_MIN = 0.5


@dataclass(frozen=True)
class AlignLine:
    idx: int
    h: str
    p: str | None
    rel: str


def parse_align_block(block: str) -> list[AlignLine] | None:
    raw = [ln.strip() for ln in block.strip().splitlines() if ln.strip()]
    if not 1 <= len(raw) <= 6:
        return None
    out = []
    for i, ln in enumerate(raw, 1):
        m = LINE_RE.match(ln)
        if not m or int(m.group(1)) != i:
            return None
        h, p, rel = m.group(2), m.group(3), m.group(4)
        if rel == "missing" and p is not None:
            return None
        if rel != "missing" and p is None:
            return None
        out.append(AlignLine(i, h, p, rel))
    return out


def rule_label(lines: list[AlignLine]) -> str:
    rels = {ln.rel for ln in lines}
    if "conflict" in rels:
        return "contradiction"
    if "missing" in rels:
        return "neutral"
    return "entailment"


def is_grounded(lines: list[AlignLine], premise: str, hypothesis: str) -> bool:
    prem, hyp = match_key(premise), match_key(hypothesis)
    for ln in lines:
        h = match_key(ln.h)
        if not h or h not in hyp:
            return False
        if ln.p is not None:
            p = match_key(ln.p)
            if not p or p not in prem:
                return False
    return True


def coverage(lines: list[AlignLine], hypothesis: str) -> float:
    toks = syllables(hypothesis)
    if not toks:
        return 0.0
    covered: set[int] = set()
    for ln in lines:
        seg = syllables(ln.h)
        n = len(seg)
        if n == 0:
            continue
        for s in range(len(toks) - n + 1):
            if toks[s:s + n] == seg:
                covered.update(range(s, s + n))
                break
    return len(covered) / len(toks)


def struct_ok(lines: list[AlignLine], premise: str, hypothesis: str, answer_label: str) -> bool:
    return (
        is_grounded(lines, premise, hypothesis)
        and coverage(lines, hypothesis) >= COVERAGE_MIN
        and rule_label(lines) == answer_label
    )
```

```python
"""Parser chặt cho output của từng format -> label, guess, phần suy luận, các dòng align, cờ format_ok. Dùng chung cho reward và eval.
"""
import re
from dataclasses import dataclass

from formats.align import AlignLine, parse_align_block

_L = "entailment|neutral|contradiction"
_ANS = rf"<answer>(?P<answer>{_L})</answer>"
_GUESS = rf"<guess>(?P<guess>{_L})</guess>\s*"
_TAGS = {
    "label": ["answer"],
    "cot": ["think", "answer"],
    "align": ["align", "answer"],
}


@dataclass
class Parsed:
    label: str | None = None
    guess: str | None = None
    reasoning: str | None = None
    align_lines: list[AlignLine] | None = None
    format_ok: bool = False


def _pattern(fmt: str, guess: bool) -> re.Pattern:
    head = _GUESS if guess else ""
    if fmt == "label":
        body = _ANS
    elif fmt == "cot":
        body = rf"<think>(?P<think>.+?)</think>\s*{_ANS}"
    else:
        body = rf"<align>\s*(?P<block>.+?)\s*</align>\s*{_ANS}"
    return re.compile(head + body, re.DOTALL)


def parse(text: str, fmt: str, guess: bool = False) -> Parsed:
    if fmt not in _TAGS:
        raise ValueError(f"format không hợp lệ: {fmt!r}")
    if guess and fmt == "label":
        raise ValueError("guess chỉ dùng cho cot/align")
    text = (text or "").strip()
    tags = _TAGS[fmt] + (["guess"] if guess else [])
    for t in tags:
        if text.count(f"<{t}>") != 1 or text.count(f"</{t}>") != 1:
            return Parsed()
    if not guess and "<guess>" in text:
        return Parsed()
    m = _pattern(fmt, guess).fullmatch(text)
    if not m:
        return Parsed()
    gd = m.groupdict()
    lines = None
    reasoning = None
    if fmt == "cot":
        reasoning = gd["think"].strip()
        if not reasoning:
            return Parsed()
    elif fmt == "align":
        reasoning = gd["block"]
        lines = parse_align_block(gd["block"])
        if lines is None:
            return Parsed()
    return Parsed(label=gd["answer"], guess=gd.get("guess"), reasoning=reasoning,
                  align_lines=lines, format_ok=True)
```

```python
"""r_acc, r_fmt, r_struct dạng hàm thuần.
"""
from formats.align import struct_ok
from formats.parser import Parsed, parse


def r_acc(p: Parsed, gold: str) -> float:
    return 1.0 if p.format_ok and p.label == gold else 0.0


def r_fmt(p: Parsed) -> float:
    return 1.0 if p.format_ok else 0.0


def r_struct(p: Parsed, fmt: str, premise: str, hypothesis: str) -> float:
    if fmt != "align" or not p.format_ok:
        return 0.0
    return 1.0 if struct_ok(p.align_lines, premise, hypothesis, p.label) else 0.0


def total_reward(text: str, fmt: str, guess: bool, gold: str, premise: str, hypothesis: str) -> float:
    p = parse(text, fmt, guess)
    a = r_acc(p, gold)
    return a + 0.2 * r_fmt(p) + 0.3 * a * r_struct(p, fmt, premise, hypothesis)
```

```python
"""Bọc thành reward_funcs cho TRL GRPOTrainer (tên r_acc, r_fmt, r_acc_struct; trọng số 1.0 / 0.2 / 0.3).
"""
from formats.parser import parse
from rewards.core import r_acc, r_fmt, r_struct

REWARD_WEIGHTS = [1.0, 0.2, 0.3]


def completion_text(c) -> str:
    if isinstance(c, str):
        return c
    if isinstance(c, list) and c and isinstance(c[-1], dict):
        return c[-1]["content"]
    raise TypeError(f"completion không hỗ trợ: {type(c)}")


def make_reward_funcs(fmt: str, guess: bool = False):
    def _parsed(completions):
        return [parse(completion_text(c), fmt, guess) for c in completions]

    def f_acc(prompts, completions, **kwargs):
        return [r_acc(p, g) for p, g in zip(_parsed(completions), kwargs["gold"])]
    f_acc.__name__ = "r_acc"

    def f_fmt(prompts, completions, **kwargs):
        return [r_fmt(p) for p in _parsed(completions)]
    f_fmt.__name__ = "r_fmt"

    def f_struct(prompts, completions, **kwargs):
        out = []
        for p, g, pr, hy in zip(_parsed(completions), kwargs["gold"],
                                kwargs["premise"], kwargs["hypothesis"]):
            out.append(r_acc(p, g) * r_struct(p, fmt, pr, hy))
        return out
    f_struct.__name__ = "r_acc_struct"

    return [f_acc, f_fmt, f_struct]
```

```python
"""Paired bootstrap 10k, McNemar, hiệu chỉnh Holm cho các so sánh đăng ký trước.
"""
import numpy as np
from scipy.stats import binomtest


def paired_bootstrap(a, b, n_boot: int = 10000, seed: int = 0) -> dict:
    d = np.asarray(a, dtype=float) - np.asarray(b, dtype=float)
    rng = np.random.default_rng(seed)
    idx = rng.integers(0, len(d), size=(n_boot, len(d)))
    means = d[idx].mean(axis=1)
    p = 2 * min((means <= 0).mean(), (means >= 0).mean())
    return {
        "diff": float(d.mean()),
        "ci_low": float(np.percentile(means, 2.5)),
        "ci_high": float(np.percentile(means, 97.5)),
        "p": float(min(p, 1.0)),
    }


def mcnemar(a, b) -> float:
    a = np.asarray(a).astype(int)
    b = np.asarray(b).astype(int)
    n10 = int(((a == 1) & (b == 0)).sum())
    n01 = int(((a == 0) & (b == 1)).sum())
    n = n10 + n01
    if n == 0:
        return 1.0
    return float(binomtest(min(n10, n01), n, 0.5).pvalue)


def holm(pvals: list[float]) -> list[float]:
    m = len(pvals)
    order = sorted(range(m), key=lambda i: pvals[i])
    adj = [0.0] * m
    running = 0.0
    for rank, i in enumerate(order):
        running = max(running, min(1.0, pvals[i] * (m - rank)))
        adj[i] = running
    return adj
```

```python
def make_dev_mini(rows: list[dict], n: int = 300, seed: int = 42) -> list[str]:
    rng = random.Random(seed)
    groups: dict[str, list[str]] = {}
    for r in sorted(rows, key=lambda r: r["id"]):
        groups.setdefault(r["label"], []).append(r["id"])
    total = len(rows)
    quota = {lab: int(n * len(ids) / total) for lab, ids in groups.items()}
    while sum(quota.values()) < n:
        lab = max(groups, key=lambda l: len(groups[l]) - quota[l])
        quota[lab] += 1
    chosen = []
    for lab in sorted(groups):
        ids = list(groups[lab])
        rng.shuffle(ids)
        chosen.extend(ids[:quota[lab]])
    return sorted(chosen)
```

```python
    "- entailment: giả thuyết suy ra đúng từ tiền đề.\n"

)

    "cot": ("Trước tiên lập luận ngắn gọn trong <think>...</think>, sau đó trả lời trong "

def build_messages(premise: str, hypothesis: str, fmt: str, guess: bool = False) -> list[dict]:
    return [{"role": "user", "content": build_prompt(premise, hypothesis, fmt, guess)}]
```

```python
"""Accuracy, macro-F1, confusion matrix, accuracy theo chủ đề, tỉ lệ parse.
"""
from common.labels import LABELS


def compute_metrics(records: list[dict]) -> dict:
    n = len(records)
    correct = sum(1 for r in records if r["pred"] == r["gold"])
    confusion: dict[str, dict[str, int]] = {}
    for r in records:
        pred = r["pred"] if r["pred"] is not None else "unparsed"
        confusion.setdefault(r["gold"], {})
        confusion[r["gold"]][pred] = confusion[r["gold"]].get(pred, 0) + 1
    f1s = []
    for lab in LABELS:
        tp = sum(1 for r in records if r["gold"] == lab and r["pred"] == lab)
        fp = sum(1 for r in records if r["gold"] != lab and r["pred"] == lab)
        fn = sum(1 for r in records if r["gold"] == lab and r["pred"] != lab)
        denom = 2 * tp + fp + fn
        f1s.append(2 * tp / denom if denom else 0.0)
    topics: dict[str, list[int]] = {}
    for r in records:
        topics.setdefault(r["topic"], []).append(int(r["pred"] == r["gold"]))
    return {
        "n": n,
        "accuracy": correct / n if n else 0.0,
        "macro_f1": sum(f1s) / len(f1s),
        "parse_rate": sum(1 for r in records if r["parse_ok"]) / n if n else 0.0,
        "confusion": confusion,
        "acc_by_topic": {t: sum(v) / len(v) for t, v in sorted(topics.items())},
    }
```

```python
from formats.align import coverage, is_grounded, rule_label
from formats.parser import parse


def build_record(row: dict, raw_output: str, fmt: str, guess: bool, n_tokens: int) -> dict:
    p = parse(raw_output, fmt, guess)
    rule = grounded = cov = None
    if fmt == "align" and p.format_ok:
        rule = rule_label(p.align_lines)
        grounded = is_grounded(p.align_lines, row["premise"], row["hypothesis"])
        cov = coverage(p.align_lines, row["hypothesis"])
    return {"id": row["id"], "dataset": row["dataset"], "topic": row["topic"],
            "gold": row["label"], "pred": p.label, "guess": p.guess,
            "parse_ok": p.format_ok, "rule_label": rule, "grounded": grounded,
            "coverage": cov, "n_tokens": n_tokens, "raw_output": raw_output}

    return {"id": row["id"], "dataset": row["dataset"], "topic": row["topic"],

            "rule_label": None, "grounded": None, "coverage": None,
```

```python
"""Suy luận greedy bằng vLLM (--language-model-only), ghi predictions_<split>.jsonl và metrics_<split>.json.
"""
import argparse
from pathlib import Path

from common.io import load_config, run_dir, write_json, write_jsonl
from data.prepare import get_rows
from eval.metrics import compute_metrics
from eval.records import build_record
from formats.prompts import build_messages


def split_allowed(split: str) -> bool:
    return split in ("dev", "dev_mini")  # test chỉ mở ở G4 qua đường riêng


def assemble_records(rows, outputs, fmt, guess) -> list[dict]:
    return [build_record(r, text, fmt, guess, n_tok) for r, (text, n_tok) in zip(rows, outputs)]


def generate(model_name, messages_list, max_new_tokens, adapter=None,
             gpu_mem=0.85, max_model_len=2048):
    from vllm import LLM, SamplingParams
    from vllm.lora.request import LoRARequest
    llm = LLM(model=model_name, language_model_only=True, dtype="bfloat16",
              enable_lora=adapter is not None, max_lora_rank=64,
              gpu_memory_utilization=gpu_mem, max_model_len=max_model_len)
    sp = SamplingParams(temperature=0.0, max_tokens=max_new_tokens)
    lora = LoRARequest("adapter", 1, str(adapter)) if adapter else None
    outs = llm.chat(messages_list, sp, lora_request=lora,
                    chat_template_kwargs={"enable_thinking": False})
    return [(o.outputs[0].text, len(o.outputs[0].token_ids)) for o in outs]


def predict_split(cfg: dict, split: str, out_dir, adapter=None) -> dict:
    assert split_allowed(split), f"split {split!r} không được phép ở giai đoạn này"
    fmt, guess = cfg["format"], cfg.get("guess", False)
    rows = get_rows(split)
    msgs = [build_messages(r["premise"], r["hypothesis"], fmt, guess) for r in rows]
    ev = cfg["eval"]
    outputs = generate(cfg["model"]["name"], msgs, ev["max_new_tokens"][fmt], adapter,
                       ev["gpu_memory_utilization"], ev["max_model_len"])
    recs = assemble_records(rows, outputs, fmt, guess)
    out_dir = Path(out_dir)
    write_jsonl(out_dir / f"predictions_{split}.jsonl", recs)
    metrics = compute_metrics(recs)
    write_json(out_dir / f"metrics_{split}.json", metrics)
    return metrics


def main(argv=None) -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--run", required=True)
    ap.add_argument("--seed", type=int, default=0)
    ap.add_argument("--split", default="dev")
    ap.add_argument("--adapter", default=None)
    ap.add_argument("--out-dir", default=None)
    a = ap.parse_args(argv)
    cfg = load_config(a.run)
    out = a.out_dir or run_dir(a.run, cfg["model"]["name"], a.seed)
    m = predict_split(cfg, a.split, out, a.adapter)
    print({k: m[k] for k in ("n", "accuracy", "macro_f1", "parse_rate")})


if __name__ == "__main__":
    main()
```

```python
from train.common import (build_sft_dataset, load_model_and_tokenizer, make_lora_config)


def pick_best(scores: dict[str, float]) -> str:
    best = max(scores.values())
    return sorted(k for k, v in scores.items() if v == best)[-1]


def train_sft(cfg: dict, seed: int, out_dir) -> list[Path]:
    from datasets import Dataset
    from trl import SFTConfig, SFTTrainer
    out_dir = Path(out_dir)
    s = cfg["sft"]

        output_dir=str(out_dir / "sft_ckpts"), seed=seed,

        lr_scheduler_type=s["scheduler"], per_device_train_batch_size=s["per_device_batch"],
        gradient_accumulation_steps=s["grad_accum"], max_length=s["max_len"],
        bf16=True, logging_steps=10, save_strategy="epoch", save_total_limit=None,
        report_to=[],
    )
    trainer = SFTTrainer(model=model, args=args, train_dataset=ds, processing_class=tok,

    trainer.train()
```

```python
"""Fine-tune encoder (CafeBERT) phân loại NLI (mode nli) hoặc chỉ hypothesis (mode hyp). Dùng cho B1 và hyp probe.
"""
import numpy as np

from common.labels import LABELS


def train_and_predict(train_rows, eval_rows, cfg: dict, mode: str, seed: int) -> np.ndarray:
    import torch
    from datasets import Dataset
    from transformers import (AutoModelForSequenceClassification, AutoTokenizer,
                              Trainer, TrainingArguments, DataCollatorWithPadding)
    assert mode in ("nli", "hyp")
    e = cfg["encoder"]
    tok = AutoTokenizer.from_pretrained(e["name"])
    model = AutoModelForSequenceClassification.from_pretrained(e["name"], num_labels=len(LABELS))
    l2i = {l: i for i, l in enumerate(LABELS)}

    def enc(rows):
        ds = Dataset.from_list([{"premise": r["premise"], "hypothesis": r["hypothesis"],
                                 "labels": l2i[r["label"]]} for r in rows])

        def fn(b):
            if mode == "nli":
                return tok(b["premise"], b["hypothesis"], truncation=True, max_length=e["max_len"])
            return tok(b["hypothesis"], truncation=True, max_length=e["max_len"])
        return ds.map(fn, batched=True, remove_columns=["premise", "hypothesis"])

    args = TrainingArguments(
        output_dir="/tmp/encoder_out", seed=seed, learning_rate=e["lr"],
        num_train_epochs=e["epochs"], per_device_train_batch_size=e["batch"],
        per_device_eval_batch_size=e["batch"] * 2, bf16=torch.cuda.is_available(),
        save_strategy="no", report_to=[], logging_steps=50,
    )
    trainer = Trainer(model=model, args=args, train_dataset=enc(train_rows),
                      data_collator=DataCollatorWithPadding(tok))
    trainer.train()
    logits = trainer.predict(enc(eval_rows)).predictions
    z = logits - logits.max(axis=1, keepdims=True)
    p = np.exp(z)
    return p / p.sum(axis=1, keepdims=True)
```

```python
"""CafeBERT hypothesis-only: 5-fold cross-fitting trên train -> p_hyp; model full-train -> nhãn hyp-easy/hyp-hard cho test ViANLI và ViNLI.

import numpy as np

from common.io import ARTIFACTS, load_config, write_jsonl


def fold_indices(n: int, k: int, seed: int):
    perm = np.random.default_rng(seed).permutation(n)
    parts = np.array_split(perm, k)
    return [(np.concatenate([parts[j] for j in range(k) if j != i]), parts[i]) for i in range(k)]


def cross_fit(train_rows: list[dict], cfg: dict, seed: int = 0) -> np.ndarray:
    from train.encoder import train_and_predict
    p = np.zeros((len(train_rows), 3))
    for tr, va in fold_indices(len(train_rows), cfg["encoder"]["folds"], seed):
        p[va] = train_and_predict([train_rows[i] for i in tr], [train_rows[i] for i in va],
                                  cfg, mode="hyp", seed=seed)
    return p


def main() -> None:
    # Chỉ phần cross-fitting trên train. Gán nhãn hyp-easy/hard cho test chờ G4 (test chưa được chạm).
    from data.prepare import get_rows

                [{"id": r["id"], "p_hyp": p[i].tolist()} for i, r in enumerate(rows)])
    print("OK", p.shape)

    main()
```

```python
"""Kiểm cổng G1: E3 > zero-shot (E0–E2) trên dev và parse E3 >= 99%; đủ E0–E3, B1, hyp probe."""
import sys

from common.io import ARTIFACTS, RESULTS, read_json

REQUIRED = ["E0", "E1", "E2", "E3", "B1"]


def evaluate_gate(metrics: dict) -> tuple[bool, list[str]]:
    lines, ok = [], True
    for rid in REQUIRED:
        if rid not in metrics:
            lines.append(f"thiếu kết quả {rid}")
            ok = False
    if not ok:
        return False, lines
    zs = max(metrics[r]["accuracy"] for r in ("E0", "E1", "E2"))
    e3 = metrics["E3"]
    better = e3["accuracy"] > zs
    parse_ok = e3["parse_rate"] >= 0.99
    lines.append(f"E3 acc {e3['accuracy']:.4f} vs best ZS {zs:.4f}: {'OK' if better else 'FAIL'}")
    lines.append(f"E3 parse {e3['parse_rate']:.4f} >= 0.99: {'OK' if parse_ok else 'FAIL'}")
    lines.append(f"B1 acc {metrics['B1']['accuracy']:.4f} (tham chiếu)")
    return better and parse_ok, lines


def main() -> None:
    metrics = {}
    for rid in REQUIRED:
        for d in sorted(RESULTS.glob(f"{rid}_*_s0")):
            f = d / "metrics_dev.json"
            if f.exists():
                metrics[rid] = read_json(f)
    ok, lines = evaluate_gate(metrics)
    probe = (ARTIFACTS / "hyp_probe" / "p_hyp_train.jsonl").exists()
    lines.append(f"hyp probe p_hyp_train.jsonl: {'OK' if probe else 'FAIL'}")
    ok = ok and probe
    print("\n".join(lines))

if __name__ == "__main__":
```

```python
"""Điểm vào duy nhất: --run E11 --seed 0 -> đọc config, gọi đúng trainer, rồi predict trên dev.

import shutil
import subprocess
import sys
import time

import yaml

from common.io import load_config, read_json, run_dir, write_json, write_jsonl


def _predict_cli(run_id, seed, split, out_dir, adapter=None):
    cmd = [sys.executable, "-m", "eval.predict", "--run", run_id, "--seed", str(seed),
           "--split", split, "--out-dir", str(out_dir)]
    if adapter:
        cmd += ["--adapter", str(adapter)]
    subprocess.run(cmd, check=True)  # tiến trình riêng để giải phóng GPU sau khi train


def _run_sft(run_id, seed, cfg, out):
    from train.sft import pick_best, train_sft
    ckpts = train_sft(cfg, seed, out)
    scores = {}
    for ck in ckpts:
        ev = out / "ckpt_eval" / ck.name
        _predict_cli(run_id, seed, "dev", ev, adapter=ck)
        scores[ck.name] = read_json(ev / "metrics_dev.json")["accuracy"]
    best = pick_best(scores)
    write_json(out / "ckpt_scores.json", {"scores": scores, "best": best})
    shutil.copytree(out / "sft_ckpts" / best, out / "best_adapter", dirs_exist_ok=True)
    for name in ("predictions_dev.jsonl", "metrics_dev.json"):
        shutil.copy(out / "ckpt_eval" / best / name, out / name)

    from train.encoder import train_and_predict

        yaml.safe_dump(cfg, f, allow_unicode=True, sort_keys=False)
```
